# tb: the ball-fed teacher (what the labels allow when the ball is seen; soft targets for distilling)
tb idsse --steps 1500 --ptr 0.5 --teach
tb metrica --steps 1500 --ptr 0.5 --teach
tb skillcorner --steps 1500 --ptr 0.5 --teach

pp idsse --steps 1500 --ptr 0.5 --ptime 2
pp metrica --steps 1500 --ptr 0.5 --ptime 2
pp skillcorner --steps 1500 --ptr 0.5 --ptime 2

sw25 metrica --steps 1500 --ptr 0.5 --provw skillcorner=0.25
sw0 metrica --steps 1500 --ptr 0.5 --provw skillcorner=0
